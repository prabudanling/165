   143→// ---------- URL index (planned sitemap, rendered by UrlDirectory) ----------
   144→
   145→export interface UrlRow {
   146→  path: string
   147→  kind: TemplateKind
   148→  schema: string
   149→  priority: number
   150→  lastmod: string
   151→}
   152→
   153→const PRIORITY: Record<TemplateKind, number> = {
   154→  person: 0.9, term: 0.9, book: 0.8, event: 0.8, place: 0.7,
   155→  institution: 0.8, sanad: 0.7, collection: 0.6, article: 0.8, faq: 0.7,
   156→}
   157→
   158→const LASTMOD_POOL = [
   159→  '2025-10-02', '2025-10-19', '2025-11-04', '2025-11-20',
   160→  '2025-12-08', '2025-12-27', '2026-01-09', '2026-01-22',
   161→]
   162→
   163→// real seed slugs appear first; the rest are clearly synthetic (no fabricated names)
   164→export function generateUrlIndex(): UrlRow[] {
   165→  const rows: UrlRow[] = []
   166→  for (const t of TEMPLATE_CATALOG) {
   167→    const entity = PREVIEW_ENTITIES[t.kind]
   168→    rows.push({
   169→      path: `${t.path}${entity.slug}`,
   170→      kind: t.kind,
   171→      schema: t.schema,
   172→      priority: PRIORITY[t.kind],
   173→      lastmod: entity.updated,
   174→    })
   175→    for (let i = 1; i < t.count; i++) {
   176→      const n = i + 1
   177→      rows.push({
   178→        path: `${t.path}${t.kind}-${String(n).padStart(6, '0')}`,
   179→        kind: t.kind,
   180→        schema: t.schema,
   181→        priority: PRIORITY[t.kind],
   182→        lastmod: LASTMOD_POOL[i % LASTMOD_POOL.length],
   183→      })
   184→    }
   185→  }
   186→  return rows
   187→}

   571→export function buildJsonLd(e: PreviewEntity): Record<string, unknown> {
   572→  const url = `${BASE_URL}${TEMPLATE_CATALOG.find((t) => t.kind === e.kind)!.path}${e.slug}`
   573→  const crumb = {
   574→    '@type': 'BreadcrumbList',
   575→    itemListElement: [
   576→      { '@type': 'ListItem', position: 1, name: 'Beranda', item: `${BASE_URL}/` },
   577→      { '@type': 'ListItem', position: 2, name: TEMPLATE_CATALOG.find((t) => t.kind === e.kind)!.label },
   578→      { '@type': 'ListItem', position: 3, name: e.name, item: url },
   579→    ],
   580→  }
   581→  let main: Record<string, unknown>
   582→  switch (e.kind) {
   583→    case 'person':
   584→      main = {
   585→        '@type': 'Person',
   586→        name: e.name,
   587→        description: e.summary,
   588→        identifier: e.globalId,
   589→        ...(e.arabic ? { alternateName: e.arabic } : {}),
   590→      }
   591→      break
   592→    case 'book':
   593→      main = {
   594→        '@type': 'Book',
   595→        name: e.name,
   596→        description: e.summary,
   597→        identifier: e.globalId,
   598→        inLanguage: 'ar',
   599→        author: { '@type': 'Person', name: 'ʿUmar ibn Muḥammad al-Suhrawardī' },
   600→      }
   601→      break
   602→    case 'term':
   603→      main = {
   604→        '@type': 'DefinedTerm',
   605→        name: e.name,
   606→        description: e.summary,
   607→        identifier: e.globalId,
   608→        inDefinedTermSet: { '@type': 'DefinedTermSet', name: 'Glossarium 165', url: `${BASE_URL}/glossary/` },
   609→      }
   610→      break
   611→    case 'event':
   612→      main = {
   613→        '@type': 'Event',
   614→        name: e.name,
   615→        description: e.summary,
   616→        identifier: e.globalId,
   617→        eventStatus: 'https://schema.org/EventScheduled',
   618→      }
   619→      break
   620→    case 'place':
   621→      main = { '@type': 'Place', name: e.name, description: e.summary, identifier: e.globalId }
   622→      break
   623→    case 'institution':
   624→      main = { '@type': 'Organization', name: e.name, description: e.summary, identifier: e.globalId }
   625→      break
   626→    case 'sanad':
   627→      main = {
   628→        '@type': 'ItemList',
   629→        name: e.name,
   630→        description: e.summary,
   631→        identifier: e.globalId,
   632→        itemListElement: (e.sanadNodes ?? []).map((n, i) => ({
   633→          '@type': 'ListItem', position: i + 1, name: n.name,
   634→          ...(n.pending ? { description: 'menunggu kolasi naskah' } : {}),
   635→        })),
   636→      }
   637→      break
   638→    case 'collection':
   639→      main = {
   640→        '@type': 'CollectionPage',
   641→        name: e.name,
   642→        description: e.summary,
   643→        identifier: e.globalId,
   644→        hasPart: [],
   645→      }
   646→      break
   647→    case 'article':
   648→      main = {
   649→        '@type': 'Article',
   650→        headline: e.name,
   651→        description: e.summary,
   652→        identifier: e.globalId,
   653→        author: { '@type': 'Organization', name: e.byline ?? 'Dewan Editorial 165' },
   654→        publisher: { '@type': 'Organization', name: '165 — TQN Qodiriah Naqsabandiyah' },
   655→      }
   656→      break
   657→    case 'faq':
   658→      main = {
   659→        '@type': 'FAQPage',
   660→        identifier: e.globalId,
   661→        mainEntity: (e.faqs ?? []).map((f) => ({
   662→          '@type': 'Question', name: f.q,
   663→          acceptedAnswer: { '@type': 'Answer', text: f.a },
   664→        })),
   665→      }
   666→      break
   667→  }
   668→  return {
   669→    '@context': 'https://schema.org',
   670→    '@graph': [
   671→      crumb,
   672→      { ...main, url, dateModified: e.updated, isPartOf: { '@type': 'WebSite', name: '165', url: `${BASE_URL}/` } },
   673→    ],
   674→  }
   675→}
